#!/usr/bin/env bash
#
# ssh_menu.sh —— Debian SSH 端口管理 · 交互式面板 (v2.8)
#
# 用途：
#   安装完成后，在命令行输入 `sshh` 即可打开本面板（安装器会注入 alias）。
#   面板提供：状态查看 / 更换端口 / 移除旧端口 / 防火墙 / 备份回滚 / 重启 / 日志 / 卸载。
#
# 用法：
#   sudo bash ssh_menu.sh            # 打开交互面板
#   sudo bash ssh_menu.sh --status   # 仅打印状态（非交互，便于脚本调用）
#   sudo bash ssh_menu.sh --help
#
# 安全设计：
#   * 需要 root 的操作统一走 as_root()，非 root 时自动用 sudo 提权
#   * 面板本身不直接改写 sshd_config，端口变更仍走 change_ssh_port.sh
#     （备份 → sshd -t 校验 → 重启 → 端口监听验证 → 失败自动回滚）
#   * 卸载时会同时清理 alias 注入块与软链，不留残留

set -uo pipefail

readonly SCRIPT_VERSION="2.8"
readonly SSHD_CONFIG="/etc/ssh/sshd_config"
readonly BACKUP_DIR="/etc/ssh/sshd_config_backups"
readonly LOG_FILE="/var/log/change_ssh_port.log"

# ---------- 颜色（NO_COLOR 或非终端时禁用） ----------
if [[ -t 1 && -z "${NO_COLOR:-}" ]]; then
    RED=$'\033[0;31m'; GREEN=$'\033[0;32m'; YELLOW=$'\033[1;33m'
    CYAN=$'\033[0;36m'; BLUE=$'\033[0;34m'; BOLD=$'\033[1m'; DIM=$'\033[2m'; NC=$'\033[0m'
else
    RED=''; GREEN=''; YELLOW=''; CYAN=''; BLUE=''; BOLD=''; DIM=''; NC=''
fi

info() { printf '%s\n' "${CYAN}[信息]${NC} $*"; }
ok()   { printf '%s\n' "${GREEN}[完成]${NC} $*"; }
warn() { printf '%s\n' "${YELLOW}[警告]${NC} $*"; }
err()  { printf '%s\n' "${RED}[错误]${NC} $*"; }

# ---------- 交互读取（curl | bash / 非终端时改从 /dev/tty 取） ----------
_tty_ok() { { : < /dev/tty; } 2>/dev/null; }

prompt() {
    local __text="$1" __ans=""
    # 提示必须单独写到 stderr。
    # 不能用 { read -p ...; } 2>/dev/null：read -p 的提示本身走 stderr，
    # 会被 2>/dev/null 一起吞掉，表现为"光标停住但没有任何提示"。
    if [[ -t 0 ]]; then
        printf '%s' "$__text" >&2
        read -r __ans
    elif _tty_ok; then
        printf '%s' "$__text" >&2
        read -r __ans < /dev/tty 2>/dev/null || return 1
    else
        return 1
    fi
    printf '%s' "$__ans"
}

# ---------- 路径解析（跟随软链，定位真实安装目录） ----------
resolve_dir() {
    local src="${BASH_SOURCE[0]}" dir
    while [[ -L "$src" ]]; do
        dir="$(cd -P "$(dirname "$src")" 2>/dev/null && pwd)"
        src="$(readlink "$src")"
        if [[ "$src" != /* ]]; then
            src="$dir/$src"
        fi
    done
    (cd -P "$(dirname "$src")" 2>/dev/null && pwd) || return 1
}

TOOL_DIR="$(resolve_dir)" || TOOL_DIR="$(pwd)"
readonly TOOL_DIR

# ---------- 提权执行 ----------
as_root() {
    if [[ $EUID -eq 0 ]]; then
        "$@"
        return $?
    fi
    if command -v sudo >/dev/null 2>&1; then
        sudo "$@"
        return $?
    fi
    err "需要 root 权限，但系统未安装 sudo，请以 root 身份重新运行。"
    return 1
}

# ---------- 状态采集 ----------
config_ports() {
    # 主配置 + Include 的子配置里的所有 Port 指令
    local pat f out=""
    {
        if [[ -f "$SSHD_CONFIG" ]]; then
            grep -E '^[[:space:]]*Port[[:space:]]+[0-9]+' "$SSHD_CONFIG" 2>/dev/null || true
        fi
        while IFS= read -r pat; do
            [[ -z "$pat" ]] && continue
            [[ "$pat" != /* ]] && pat="/etc/ssh/$pat"
            for f in $pat; do
                if [[ -f "$f" ]]; then
                    grep -E '^[[:space:]]*Port[[:space:]]+[0-9]+' "$f" 2>/dev/null || true
                fi
            done
        done < <(grep -E '^[[:space:]]*Include[[:space:]]+' "$SSHD_CONFIG" 2>/dev/null || true)
    } | awk '{print $2}' | sort -un
}

effective_ports() {
    # sshd -T 需要 root；失败时回退到解析配置文件
    local out=""
    if [[ $EUID -eq 0 ]] && command -v sshd >/dev/null 2>&1; then
        out="$(sshd -T 2>/dev/null | awk '/^port /{print $2}' | sort -un || true)"
    fi
    if [[ -z "$out" ]]; then
        out="$(config_ports)"
    fi
    if [[ -z "$out" ]]; then
        out="22"
    fi
    printf '%s' "$out"
}

port_listening() {
    local port="$1"
    if command -v ss >/dev/null 2>&1; then
        ss -tln 2>/dev/null | awk '{print $4}' | grep -qE "[:.]${port}\$"
    elif command -v netstat >/dev/null 2>&1; then
        netstat -tln 2>/dev/null | awk '{print $4}' | grep -qE "[:.]${port}\$"
    else
        return 1
    fi
}

svc_status() {
    local state="unknown" detail=""
    if command -v systemctl >/dev/null 2>&1; then
        state="$(systemctl is-active ssh 2>/dev/null || true)"
        if [[ -z "$state" || "$state" == "inactive" || "$state" == "failed" ]]; then
            state="$(systemctl is-active sshd 2>/dev/null || true)"
        fi
        detail="$(systemctl show ssh -p SubState --value 2>/dev/null || true)"
        if [[ -z "$detail" ]]; then
            detail="$(systemctl show sshd -p SubState --value 2>/dev/null || true)"
        fi
    elif command -v service >/dev/null 2>&1; then
        if service ssh status >/dev/null 2>&1; then
            state="active"
        fi
    fi
    [[ -z "$state" ]] && state="unknown"
    if [[ -n "$detail" ]]; then
        printf '%s %s' "$state" "$detail"
    else
        printf '%s' "$state"
    fi
}

firewall_info() {
    local kind="none" extra=""
    if command -v ufw >/dev/null 2>&1; then
        kind="ufw"
        if [[ $EUID -eq 0 ]]; then
            extra="$(ufw status 2>/dev/null | awk '/^Status:/{print $2}' || true)"
        fi
    elif command -v nft >/dev/null 2>&1; then
        kind="nftables"
    elif command -v iptables >/dev/null 2>&1; then
        kind="iptables"
    fi
    printf '%s %s' "$kind" "$extra"
}

config_valid() {
    if [[ $EUID -ne 0 ]] || ! command -v sshd >/dev/null 2>&1; then
        printf 'unknown'
        return
    fi
    if sshd -t >/dev/null 2>&1; then
        printf '通过'
    else
        printf '失败'
    fi
}

# 列出备份文件（按修改时间倒序，只输出文件名）
list_backups() {
    local dir="$1"
    [[ -d "$dir" ]] || return 0
    if find "$dir" -maxdepth 1 -type f -name '*.bak' -printf '%T@ %f\n' >/dev/null 2>&1; then
        find "$dir" -maxdepth 1 -type f -name '*.bak' -printf '%T@ %f\n' 2>/dev/null \
            | sort -rn | cut -d' ' -f2-
    else
        # 降级路径：GNU find 不支持 -printf 时退回 ls
        # shellcheck disable=SC2010,SC2012
        ls -1t "$dir" 2>/dev/null | grep -E '\.bak$' || true
    fi
}

last_backup() {
    local f
    f="$(list_backups "$BACKUP_DIR" | awk 'NR==1{print; exit}')"
    if [[ -n "$f" ]]; then
        printf '%s' "$f"
    else
        printf '无'
    fi
}

host_summary() {
    local os="Linux"
    if [[ -r /etc/os-release ]]; then
        os="$(awk -F= '/^PRETTY_NAME=/{gsub(/"/,"",$2); print $2; exit}' /etc/os-release 2>/dev/null || true)"
    fi
    [[ -z "$os" ]] && os="Linux"
    printf '%s@%s · %s' "${USER:-root}" "$(hostname 2>/dev/null || echo localhost)" "$os"
}

# ---------- 面板渲染 ----------
draw_header() {
    local ports listen svc fw valid backup
    ports="$(effective_ports)"
    listen=""
    local p
    while IFS= read -r p; do
        [[ -z "$p" ]] && continue
        if port_listening "$p"; then
            listen+=" ${GREEN}${p} ✓${NC}"
        else
            listen+=" ${YELLOW}${p} ✗${NC}"
        fi
    done <<< "$ports"

    svc="$(svc_status)"
    fw="$(firewall_info)"
    valid="$(config_valid)"
    backup="$(last_backup)"

    printf '%s\n' "${CYAN}╔══════════════════════════════════════════════════════════════╗${NC}"
    printf '%s\n' "${CYAN}║${NC}   ${BOLD}Debian SSH · 端口管理面板${NC}                ${BLUE}v${SCRIPT_VERSION}${NC}          ${CYAN}║${NC}"
    printf '%s\n' "${CYAN}╚══════════════════════════════════════════════════════════════╝${NC}"
    printf '  %s%-9s%s: %s\n' "$DIM" "主机" "$NC" "$(host_summary)"
    printf '  %s%-9s%s: %s\n' "$DIM" "当前端口" "$NC" "$(printf '%s' "$ports" | tr '\n' ' ')"
    printf '  %s%-9s%s:%s\n' "$DIM" "端口监听" "$NC" "$listen"
    printf '  %s%-9s%s: %s\n' "$DIM" "服务状态" "$NC" "$svc"
    printf '  %s%-9s%s: %s\n' "$DIM" "防火墙" "$NC" "$fw"
    printf '  %s%-9s%s: %s\n' "$DIM" "配置校验" "$NC" "$valid"
    printf '  %s%-9s%s: %s\n' "$DIM" "最近备份" "$NC" "$backup"
    printf '%s\n' "${DIM}  ────────────────────────────────────────────────────────────${NC}"
}

draw_menu() {
    printf '  %s1)%s 查看 SSH 状态详情\n' "$BOLD" "$NC"
    printf '  %s2)%s 更改 SSH 端口\n' "$BOLD" "$NC"
    printf '  %s3)%s 移除旧端口（过渡期收尾）\n' "$BOLD" "$NC"
    printf '  %s4)%s 备份与回滚\n' "$BOLD" "$NC"
    printf '  %s5)%s 防火墙规则\n' "$BOLD" "$NC"
    printf '  %s6)%s 重启 SSH 服务\n' "$BOLD" "$NC"
    printf '  %s7)%s 查看操作日志\n' "$BOLD" "$NC"
    printf '  %s8)%s 卸载本工具\n' "$BOLD" "$NC"
    printf '  %s0)%s 退出\n' "$BOLD" "$NC"
    printf '%s\n' "${DIM}  ────────────────────────────────────────────────────────────${NC}"
}

clear_screen() {
    printf '\033[H\033[2J\033[3J'
}

pause_enter() {
    local _
    if [[ -t 0 ]]; then
        read -r -p "  按 Enter 返回菜单..." _ || true
    else
        sleep 1
    fi
}

# ---------- 具体功能 ----------
show_detail() {
    printf '\n%s\n' "${BOLD}── SSH 状态详情 ──${NC}"
    printf '  配置文件    : %s\n' "$SSHD_CONFIG"
    printf '  备份目录    : %s\n' "$BACKUP_DIR"
    printf '  日志文件    : %s\n' "$LOG_FILE"

    local ports
    ports="$(effective_ports)"
    printf '  生效端口    : %s\n' "$(printf '%s' "$ports" | tr '\n' ' ')"

    printf '  监听明细    :\n'
    if command -v ss >/dev/null 2>&1; then
        # 端口需整体成组匹配，否则 22 会匹配到 2222 这类行
        local port_pat
        port_pat="$(printf '%s' "$ports" | tr '\n' '|')"
        port_pat="${port_pat%|}"
        ss -tlnp 2>/dev/null | grep -E "sshd|:(${port_pat})[[:space:]]" \
            || printf '    （未找到 sshd 监听记录）\n'
    else
        printf '    （ss 不可用）\n'
    fi

    printf '  配置中的 Port 指令:\n'
    if [[ -f "$SSHD_CONFIG" ]]; then
        grep -nE '^[[:space:]]*Port[[:space:]]+[0-9]+' "$SSHD_CONFIG" 2>/dev/null \
            | while IFS= read -r line; do printf '    %s\n' "$line"; done
    else
        printf '    （配置文件不存在）\n'
    fi

    printf '  防火墙:\n'
    if command -v ufw >/dev/null 2>&1; then
        as_root ufw status 2>/dev/null | while IFS= read -r line; do printf '    %s\n' "$line"; done
    elif command -v iptables >/dev/null 2>&1; then
        as_root iptables -S 2>/dev/null | grep -E "dport|ACCEPT" | while IFS= read -r line; do printf '    %s\n' "$line"; done
    else
        printf '    （未检测到常见防火墙工具）\n'
    fi
    printf '\n'
}

do_change_port() {
    printf '\n'
    if [[ ! -f "$TOOL_DIR/change_ssh_port.sh" ]]; then
        err "未找到 change_ssh_port.sh（安装目录：$TOOL_DIR）"
        return 1
    fi
    as_root bash "$TOOL_DIR/change_ssh_port.sh"
}

do_remove_old_port() {
    printf '\n'
    if [[ ! -f "$TOOL_DIR/remove_old_ssh_port.sh" ]]; then
        err "未找到 remove_old_ssh_port.sh（安装目录：$TOOL_DIR）"
        return 1
    fi
    as_root bash "$TOOL_DIR/remove_old_ssh_port.sh"
}

do_backup() {
    printf '\n%s\n' "${BOLD}── 备份与回滚 ──${NC}"
    local files=()
    mapfile -t files < <(list_backups "$BACKUP_DIR")
    if [[ ${#files[@]} -eq 0 ]]; then
        warn "备份目录为空：$BACKUP_DIR"
        return 0
    fi

    local i=1
    local f
    for f in "${files[@]}"; do
        printf '  %s%2d)%s %s  (%s)\n' "$BOLD" "$i" "$NC" "$f" "$(date -r "$BACKUP_DIR/$f" '+%F %T' 2>/dev/null || echo '-')"
        i=$((i + 1))
    done

    local ans
    ans="$(prompt "$(echo -e "${CYAN}输入要回滚的编号（直接回车返回）：${NC}")")" || ans=""
    if [[ -z "$ans" ]]; then
        return 0
    fi
    if ! [[ "$ans" =~ ^[0-9]+$ ]] || (( ans < 1 || ans > ${#files[@]} )); then
        err "编号无效。"
        return 1
    fi

    local target="${files[$((ans - 1))]}"
    warn "将用备份回滚当前配置：$target"
    local c
    c="$(prompt "$(echo -e "${CYAN}确认回滚？(y/N) ${NC}")")" || c=""
    if [[ "$c" != [yY] ]]; then
        info "已取消。"
        return 0
    fi

    # 覆盖前先留存当前配置：若备份本身是坏的，可立即恢复，避免把可用配置覆盖掉
    local pre="$SSHD_CONFIG.pre-rollback.$$"
    as_root cp -p "$SSHD_CONFIG" "$pre" 2>/dev/null || true
    if ! as_root cp -p "$BACKUP_DIR/$target" "$SSHD_CONFIG"; then
        err "回滚失败：无法写入 $SSHD_CONFIG"
        as_root rm -f "$pre" 2>/dev/null || true
        return 1
    fi
    if as_root sshd -t >/dev/null 2>&1; then
        as_root rm -f "$pre" 2>/dev/null || true
        if as_root systemctl restart ssh 2>/dev/null \
            || as_root service ssh restart 2>/dev/null; then
            ok "已回滚并重启 SSH 服务。"
        else
            warn "已回滚，但服务重启失败，请手动检查：systemctl status ssh"
        fi
    else
        # 备份本身是坏的：必须恢复回滚前的配置，否则会把可用配置覆盖成坏的
        as_root cp -p "$pre" "$SSHD_CONFIG" 2>/dev/null || true
        as_root rm -f "$pre" 2>/dev/null || true
        err "该备份的配置语法校验失败，已恢复回滚前的配置。"
        err "请手动检查备份文件：$BACKUP_DIR/$target"
        return 1
    fi
}

do_firewall() {
    printf '\n%s\n' "${BOLD}── 防火墙规则 ──${NC}"
    local ports
    ports="$(effective_ports)"

    if command -v ufw >/dev/null 2>&1; then
        printf '  ufw 状态:\n'
        as_root ufw status 2>/dev/null | while IFS= read -r line; do printf '    %s\n' "$line"; done
    elif command -v nft >/dev/null 2>&1; then
        printf '  nftables 规则:\n'
        as_root nft list ruleset 2>/dev/null | head -n 30 | while IFS= read -r line; do printf '    %s\n' "$line"; done
    elif command -v iptables >/dev/null 2>&1; then
        printf '  iptables 规则:\n'
        as_root iptables -S 2>/dev/null | while IFS= read -r line; do printf '    %s\n' "$line"; done
    else
        warn "未检测到 ufw / nftables / iptables。"
        return 0
    fi

    local ans
    ans="$(prompt "$(echo -e "${CYAN}是否为当前端口 ${BOLD}${ports//$'\n'/ }${NC} 补充放行规则？(y/N) ${NC}")")" || ans=""
    if [[ "$ans" == [yY] ]]; then
        local p
        while IFS= read -r p; do
            [[ -z "$p" ]] && continue
            if command -v ufw >/dev/null 2>&1; then
                if as_root ufw allow "${p}/tcp" >/dev/null 2>&1; then
                    ok "ufw 已放行 ${p}/tcp"
                else
                    warn "ufw 放行 ${p}/tcp 失败"
                fi
            elif command -v iptables >/dev/null 2>&1; then
                if as_root iptables -C INPUT -p tcp --dport "$p" -j ACCEPT 2>/dev/null; then
                    ok "iptables 规则已存在（${p}/tcp）"
                elif as_root iptables -A INPUT -p tcp --dport "$p" -j ACCEPT 2>/dev/null; then
                    ok "iptables 已放行 ${p}/tcp（重启可能失效，建议持久化）"
                else
                    warn "iptables 放行 ${p}/tcp 失败"
                fi
            elif command -v nft >/dev/null 2>&1; then
                # nft 的规则位置因发行版而异，不自动改，避免把规则插错表/链
                warn "nftables 需手动放行 ${p}/tcp："
                printf '      nft add rule inet filter input tcp dport %s accept\n' "$p"
            fi
        done <<< "$ports"
    fi
}

do_restart() {
    printf '\n'
    warn "重启 SSH 服务会短暂中断连接（当前会话不受影响）。"
    local c
    c="$(prompt "$(echo -e "${CYAN}确认重启？(y/N) ${NC}")")" || c=""
    if [[ "$c" != [yY] ]]; then
        info "已取消。"
        return 0
    fi
    if as_root systemctl restart ssh 2>/dev/null; then
        ok "ssh 已重启。"
    elif as_root systemctl restart sshd 2>/dev/null; then
        ok "sshd 已重启。"
    elif as_root service ssh restart 2>/dev/null; then
        ok "ssh 已重启（service）。"
    else
        err "重启失败，请手动检查：systemctl status ssh"
        return 1
    fi
    sleep 1
    local ports p
    ports="$(effective_ports)"
    while IFS= read -r p; do
        [[ -z "$p" ]] && continue
        if port_listening "$p"; then
            ok "端口 $p 正在监听。"
        else
            warn "端口 $p 未检测到监听，请检查：ss -tlnp | grep sshd"
        fi
    done <<< "$ports"
}

    local c
    c="$(prompt "$(echo -e "${CYAN}确认卸载？(y/N) ${NC}")")" || c=""
    if [[ "$c" != [yY] ]]; then
        info "已取消。"
        return 0
    fi

    # 清理 alias 注入块
    # sudo 下 $HOME 是 /root，需反查 SUDO_USER 的真实 home，否则普通用户的配置清不掉
    local target_home="$HOME" h
    if [[ -n "${SUDO_USER:-}" && "$SUDO_USER" != "root" ]]; then
        if command -v getent >/dev/null 2>&1; then
            h="$(getent passwd "$SUDO_USER" 2>/dev/null | awk -F: '{print $6; exit}')" || h=""

        fi

    if [[ "$target_home" != "/root" && -d /root ]]; then
        homes+=("/root")
    fi
    for home in "${homes[@]}"; do
        for rc in "$home/.bashrc" "$home/.zshrc"; do
            [[ -f "$rc" ]] || continue
            if grep -q 'ssh-port-tool' "$rc" 2>/dev/null; then
                if sed -i '/^# >>> ssh-port-tool >>>$/,/^# <<< ssh-port-tool <<<$/d' "$rc" 2>/dev/null; then
                    ok "已清理别名配置：$rc"
                else
                    warn "清理别名失败，请手动编辑：$rc"
                fi
            fi
        done
    done

    # 清理软链（与安装器创建的 4 个保持一致）
    local link
    for link in /usr/local/bin/ssh-port /usr/local/bin/ssh-port-menu \
                /usr/local/bin/ssh-port-change /usr/local/bin/ssh-port-remove; do

    printf '  安装目录将在本脚本退出后删除：%s\n' "$TOOL_DIR"

}

#!/usr/bin/env bash
#
# ssh-port.sh —— Debian SSH 端口工具 · 命令包装器 (v2.8)
#
# 行为约定：
#
#   ssh-port                    → 打开交互式管理面板（无参数）
#   ssh-port user@host          → 完全等价于原生 ssh，转发所有参数
#   ssh-port -p 2222 root@host  → 同上，带端口/选项也照常转发
#
# 安装器会向 ~/.bashrc、~/.zshrc 注入：
#   alias sshh='ssh-port'
# 于是：
#   sshh                → 打开管理面板
#   sshh root@1.2.3.4   → 转发到原生 SSH 客户端
#
# 设计要点：刻意不占用 ssh 这个名字，避免与系统命令冲突。
# 系统自带的 ssh 始终原封不动，ssh user@host 行为与安装前完全一致。
#
# 如需绕过别名直接运行面板：ssh-port 或 ssh-port-menu

set -uo pipefail

readonly SCRIPT_VERSION="2.8"

# ---------- 定位真实安装目录（跟随软链） ----------
resolve_dir() {
    local src="${BASH_SOURCE[0]}" dir
    while [[ -L "$src" ]]; do
        dir="$(cd -P "$(dirname "$src")" 2>/dev/null && pwd)"
        src="$(readlink "$src")"
        if [[ "$src" != /* ]]; then
            src="$dir/$src"
        fi
    done
    (cd -P "$(dirname "$src")" 2>/dev/null && pwd) || return 1
}

TOOL_DIR="$(resolve_dir)" || TOOL_DIR="$(pwd)"

# ---------- 定位原生 ssh 客户端 ----------
# 依次尝试常见路径；跳过指向本工具自身的软链，避免无限递归
find_real_ssh() {
    local candidate resolved
    for candidate in /usr/bin/ssh /bin/ssh /usr/local/bin/ssh /usr/local/sbin/ssh; do
        [[ -x "$candidate" ]] || continue
        if [[ -L "$candidate" ]]; then
            resolved="$(readlink -f "$candidate" 2>/dev/null || true)"
            # 指向本工具自身的链接 → 跳过
            if [[ "$resolved" == "$TOOL_DIR"/* ]]; then
                continue
            fi
        fi
        printf '%s' "$candidate"
        return 0
    done
    # 兜底：在 PATH 中查找，但排除自身目录
    local IFS=':'
    local p
    for p in $PATH; do
        [[ -z "$p" ]] && continue
        candidate="$p/ssh"
        [[ -x "$candidate" ]] || continue
        resolved="$(readlink -f "$candidate" 2>/dev/null || true)"
        if [[ "$resolved" == "$TOOL_DIR"/* ]]; then
            continue
        fi
        printf '%s' "$candidate"
        return 0
    done
    return 1
}

    cat <<EOF
Debian SSH 端口工具包装器 v${SCRIPT_VERSION}

用法:
  ssh-port                 打开交互式管理面板（无参数时）
  ssh-port <ssh参数...>    完全等价于原生 ssh（原样透传）

说明:
  安装器已注入 alias sshh='ssh-port'，因此终端里输入 sshh 即可打开面板；
  系统自带的 ssh 命令完全不受影响，ssh user@host 行为与安装前一致。

  本包装器自有选项（不会与 ssh 参数冲突）:
    --menu           打开管理面板
    --tool-version   显示本工具版本
    --tool-help      显示本帮助

  直接运行面板（不依赖别名）:
    ssh-port
    ssh-port-menu
EOF
}

main() {
    # 无参数 → 打开交互面板
    if [[ $# -eq 0 ]]; then
        local menu="$TOOL_DIR/ssh_menu.sh"
        if [[ ! -f "$menu" ]]; then
            printf '[错误] 未找到面板脚本: %s\n' "$menu" >&2
            return 1
        fi
        exec bash "$menu" --menu
    fi

    # 仅识别本工具自有选项；其余一律原样转发，绝不截获 ssh 的参数
    # （例如 ssh -V 必须显示 OpenSSH 版本，而不是本工具版本）
    case "$1" in
        --menu)
            local menu="$TOOL_DIR/ssh_menu.sh"
            if [[ ! -f "$menu" ]]; then
                printf '[错误] 未找到面板脚本: %s\n' "$menu" >&2
                return 1
            fi
            exec bash "$menu" --menu
            ;;
        --tool-version)
            printf 'ssh-port.sh v%s\n' "$SCRIPT_VERSION"
            return 0 ;;
        --tool-help)
            usage
            return 0 ;;
    esac

    # 带参数 → 转发原生 ssh
    local real
    real="$(find_real_ssh)" || {
        printf '[错误] 未找到原生 ssh 客户端，请确认已安装 openssh-client。\n' >&2
        return 127
    }
    exec "$real" "$@"
}

main "$@"

#!/usr/bin/env bash
#
# change_ssh_port.sh —— Debian 交互式/半自动 SSH 端口更换脚本 (v2.8)
#
# 相对 v1 的改进：
#   + 支持命令行参数（-p 端口 / -y 自动确认 / --keep-old / --no-firewall / -n 预演）
#   + 检测 sshd_config 中的 Include 指令，避免 sshd_config.d/*.conf 里的 Port 覆盖
#   + 写入日志 /var/log/change_ssh_port.log
#   + Port 指令写入文件顶部，规避 Match 块导致的指令归属错误
#   + 重启后自动验证新端口是否真正在监听
#   + 全程 trap 兜底回滚，异常自动还原备份
#
# 用法：
#   sudo bash change_ssh_port.sh                     # 纯交互
#   sudo bash change_ssh_port.sh -p 2222             # 指定端口，其余交互
#   sudo bash change_ssh_port.sh -p 2222 -y          # 全自动（默认保留旧端口）
#   sudo bash change_ssh_port.sh -p 2222 -y --no-keep-old --firewall
#   sudo bash change_ssh_port.sh -p 2222 -n          # 预演，只显示将要做的改动

readonly SSHD_CONFIG="/etc/ssh/sshd_config"

# ---------- 颜色 ----------
RED=$'\033[0;31m'; GREEN=$'\033[0;32m'; YELLOW=$'\033[1;33m'
CYAN=$'\033[0;36m'; BOLD=$'\033[1m'; NC=$'\033[0m'

info() { echo -e "${CYAN}[信息]${NC} $*"; }
ok()   { echo -e "${GREEN}[完成]${NC} $*"; }
warn() { echo -e "${YELLOW}[警告]${NC} $*"; }
err()  { echo -e "${RED}[错误]${NC} $*"; }

# 交互读取：stdin 不是终端时（如 curl | bash 管道执行）改从 /dev/tty 读取
# 用法：ans="$(prompt "提示文字")" || ans=""
# 用子 shell 探测 /dev/tty 是否真可用，避免 "No such device or address" 噪音
_tty_ok() { { : < /dev/tty; } 2>/dev/null; }

prompt() {
    local __text="$1" __ans=""
    # 提示必须单独写到 stderr。
    # 不能用 { read -p ...; } 2>/dev/null：read -p 的提示本身走 stderr，
    # 会被 2>/dev/null 一起吞掉，表现为"光标停住但没有任何提示"。
    if [[ -t 0 ]]; then
        printf '%s' "$__text" >&2
        read -r __ans
    elif _tty_ok; then
        printf '%s' "$__text" >&2
        read -r __ans < /dev/tty 2>/dev/null || return 1
    else
        return 1
    fi
    printf '%s' "$__ans"
}

log() {
    local ts; ts="$(date '+%F %T')"
    echo "[$ts] [v${SCRIPT_VERSION}] $*" >> "$LOG_FILE" 2>/dev/null || true
}

# 重启 SSH 服务：不同发行版服务名可能是 ssh 或 sshd
restart_ssh() {
    systemctl restart ssh  2>/dev/null && return 0
    systemctl restart sshd 2>/dev/null && return 0
    service ssh restart    2>/dev/null && return 0
    service sshd restart   2>/dev/null && return 0
    return 1
}

require_root() {
    if [[ $EUID -ne 0 ]]; then
        err "请使用 root 权限运行（例如：sudo bash $0）"
        exit 1
    fi
}

# ---------- 端口工具 ----------
get_config_ports() {
    grep -E '^[[:space:]]*Port[[:space:]]+' "$SSHD_CONFIG" 2>/dev/null \
        | awk '{print $2}' | sort -un || true
}

get_effective_ports() {
    sshd -T 2>/dev/null | awk '/^port /{print $2}' | sort -un || true
}

port_listening() {
    local port="$1"
    # 第 4 列为 Local Address:Port；表头该列不含端口号，故无需跳过首行
    if command -v ss >/dev/null 2>&1; then
        ss -tln 2>/dev/null | awk '{print $4}' | grep -qE "[:.]${port}\$"
    elif command -v netstat >/dev/null 2>&1; then
        netstat -tln 2>/dev/null | awk '{print $4}' | grep -qE "[:.]${port}\$"
    else
        return 1
    fi
}

validate_port() {
    local port="$1" current_ports="$2"
    [[ "$port" =~ ^[0-9]+$ ]] || { err "端口必须是数字。"; return 1; }
    (( port >= 1 && port <= 65535 )) || { err "端口必须在 1-65535 之间。"; return 1; }
    if (( port < 1024 )); then
        warn "端口 < 1024 属特权端口，建议改用 1024-65535。"
    fi
    if grep -qw "$port" <<< "$current_ports"; then
        err "端口 $port 已是当前 SSH 端口，请换一个。"
        return 1
    fi
    if port_listening "$port"; then
        err "端口 $port 已被本机其他服务占用，请换一个。"
        return 1
    fi
    return 0
}

# ---------- Include 处理 ----------
handle_includes() {
    local include_patterns
    include_patterns="$(grep -E '^[[:space:]]*Include[[:space:]]+' "$SSHD_CONFIG" 2>/dev/null | awk '{print $2}' || true)"
    [[ -z "$include_patterns" ]] && return 0

    local files=() pat f
    while IFS= read -r pat; do
        [[ -z "$pat" ]] && continue
        [[ "$pat" != /* ]] && pat="/etc/ssh/$pat"
        for f in $pat; do
            [[ -f "$f" ]] && files+=("$f")
        done
    done <<< "$include_patterns"

    [[ ${#files[@]} -eq 0 ]] && return 0

    local hits=()
    for f in "${files[@]}"; do
        if grep -qE '^[[:space:]]*Port[[:space:]]+[0-9]+' "$f" 2>/dev/null; then
            hits+=("$f")
        fi
    done

    if [[ ${#hits[@]} -eq 0 ]]; then
        info "已检查 Include 配置，未发现额外 Port 指令。"
        return 0
    fi

    warn "以下被 Include 的配置文件中也定义了 Port，若不处理会继续监听旧端口："
    for f in "${hits[@]}"; do
        echo -e "    ${YELLOW}$f${NC}: $(grep -E '^[[:space:]]*Port[[:space:]]+' "$f" | tr '\n' ' ')"
    done

    local ans
    if [[ $ASSUME_YES -eq 1 ]]; then
        ans="y"
    else
        ans="$(prompt "$(echo -e "${CYAN}是否将这些 Include 文件中的 Port 指令注释掉？(Y/n) ${NC}")")" || ans=""
    fi
    case "$ans" in
        [nN]) warn "保留 Include 中的 Port，旧端口可能仍生效，请知悉。"; return 0 ;;
    esac

    for f in "${hits[@]}"; do
        local bak
        bak="${f}.bak.$(date +%Y%m%d-%H%M%S)"
        cp -p "$f" "$bak"
        INCLUDE_BACKUPS+=("${f}|${bak}")
        sed -i -E 's|^([[:space:]]*)(Port[[:space:]]+)|\1#\2|' "$f"
        ok "已注释 $f 中的 Port 指令（备份：$bak）"
        log "commented Port in include: $f (backup: $bak)"
    done
}

# ---------- 防火墙 ----------
configure_firewall() {
    local port="$1"
    local ans
    if [[ -n "$DO_FIREWALL" ]]; then
        ans="$DO_FIREWALL"
    elif [[ $ASSUME_YES -eq 1 ]]; then
        ans="yes"
    else
        echo
        info "放行新端口 $port 的防火墙规则（否则可能连不上）。"
        ans="$(prompt "$(echo -e "${CYAN}现在配置防火墙放行 $port 吗？(Y/n) ${NC}")")" || ans=""
    fi
    case "$ans" in
        no|[nN]) warn "已跳过防火墙配置，请手动放行端口 $port。"; return ;;
    esac

    if command -v ufw >/dev/null 2>&1; then
        if ufw allow "${port}/tcp" >/dev/null 2>&1; then
            ok "ufw 已放行 $port/tcp"
        else
            err "ufw 放行失败，请手动检查。"
        fi
    elif command -v nft >/dev/null 2>&1; then
        warn "检测到 nftables，请手动确认放行 $port/tcp（nft add rule inet filter input tcp dport $port accept）。"
    elif command -v iptables >/dev/null 2>&1; then
        if iptables -C INPUT -p tcp --dport "$port" -j ACCEPT 2>/dev/null; then
            ok "iptables 规则已存在（$port/tcp）"
        elif iptables -A INPUT -p tcp --dport "$port" -j ACCEPT 2>/dev/null; then
            ok "iptables 已放行 $port/tcp（重启可能失效，建议持久化）"
        else
            err "iptables 放行失败，请手动检查。"
        fi
    else
        warn "未检测到常见防火墙工具，请自行确认端口可达。"
    fi
}

# ---------- 配置写入 ----------
BACKUP_FILE=""
CHANGED=0
TMP_BODY=""
# 被注释掉 Port 的 Include 文件：每项格式 "源文件|备份文件"
INCLUDE_BACKUPS=()

on_exit() {
    local rc=$?
    # 先兜底回滚
    if [[ $rc -ne 0 && $CHANGED -eq 1 && -n "$BACKUP_FILE" && -f "$BACKUP_FILE" ]]; then
        err "检测到异常，正在回滚 sshd_config ..."
        cp -p "$BACKUP_FILE" "$SSHD_CONFIG" 2>/dev/null || true
        # Include 文件同样要还原，否则改动会残留
        local pair src bak
        for pair in "${INCLUDE_BACKUPS[@]}"; do
            src="${pair%%|*}"
            bak="${pair#*|}"
            if [[ -f "$bak" ]]; then
                cp -p "$bak" "$src" 2>/dev/null || true
                log "restored include: $src"
            fi
        done
        restart_ssh || true
        err "已回滚到修改前状态（含 Include 文件）。"
        log "ROLLBACK triggered, rc=$rc"
    fi
    # 再清理临时文件
    [[ -n "${TMP_BODY:-}" && -f "${TMP_BODY:-}" ]] && rm -f "$TMP_BODY"
}
trap on_exit EXIT

apply_config() {
    local new_port="$1" old_port="$2" keep_old="$3"

    if [[ $DRY_RUN -eq 1 ]]; then
        info "[预演] 将写入的 Port 指令："
        [[ "$keep_old" == "yes" ]] && echo -e "    ${BOLD}Port $old_port${NC}   (保留旧端口)"
        echo -e "    ${BOLD}Port $new_port${NC}"
        info "[预演] 其余配置保持不变；不写文件、不重启服务。"
        return 0
    fi

    TMP_BODY="$(mktemp)"
    # 删除主配置中所有 Port 指令（含被注释的），避免重复/冲突
    grep -vE '^[[:space:]]*#?[[:space:]]*Port[[:space:]]+[0-9]+[[:space:]]*(#.*)?$' "$SSHD_CONFIG" > "$TMP_BODY" || true

    local new_content; new_content="$(mktemp)"
    {
        echo "# === change_ssh_port.sh v${SCRIPT_VERSION} @ $(date '+%F %T') ==="
        [[ "$keep_old" == "yes" ]] && echo "Port $old_port"
        echo "Port $new_port"
        echo ""
        cat "$TMP_BODY"
    } > "$new_content"

    chmod --reference="$SSHD_CONFIG" "$new_content" 2>/dev/null || chmod 644 "$new_content"
    chown --reference="$SSHD_CONFIG" "$new_content" 2>/dev/null || true

    mv "$new_content" "$SSHD_CONFIG"
    CHANGED=1
}

# ---------- 主流程 ----------
main() {
    echo -e "${BOLD}==================================================${NC}"
    echo -e "${BOLD}     Debian SSH 端口更换脚本  v${SCRIPT_VERSION}${NC}"
    echo -e "${BOLD}==================================================${NC}"
    echo

    require_root

    [[ -f "$SSHD_CONFIG" ]] || { err "未找到 $SSHD_CONFIG"; exit 1; }

    # 非交互环境且未指定端口时，直接提示，避免在 read 循环里空转
    if [[ -z "$NEW_PORT" && ! -t 0 ]] && ! _tty_ok; then
        err "当前为非交互环境，请用 -p 指定端口，例如：bash $0 -p 2222 -y"
        exit 1
    fi

    local old_port current_ports
    old_port="$(get_config_ports | awk 'NR==1{print; exit}')"
    old_port="${old_port:-22}"
    current_ports="$(get_effective_ports)"
    [[ -z "$current_ports" ]] && current_ports="$old_port"

    info "当前生效 SSH 端口：${BOLD}${current_ports//$'\n'/ }${NC}"
    log "start: current ports=$(echo "$current_ports" | tr '\n' ' ')"

    # 新端口：参数优先，否则交互
    if [[ -n "$NEW_PORT" ]]; then
        validate_port "$NEW_PORT" "$current_ports" || { err "参数 -p 指定的端口无效"; exit 1; }
    else
        while true; do
            NEW_PORT="$(prompt "$(echo -e "${CYAN}请输入新的 SSH 端口 (1-65535)：${NC}")")" || NEW_PORT=""
            validate_port "$NEW_PORT" "$current_ports" && break
        done
    fi

    # 保留旧端口策略
    if [[ -z "$KEEP_OLD" ]]; then
        if [[ $ASSUME_YES -eq 1 ]]; then
            KEEP_OLD="yes"
        else
            echo
            warn "建议过渡期保留旧端口（$old_port），验证新端口能登录后再关闭。"
            a="$(prompt "$(echo -e "${CYAN}临时保留旧端口 $old_port 吗？(Y/n) ${NC}")")" || a=""
            case "$a" in [nN]) KEEP_OLD="no" ;; *) KEEP_OLD="yes" ;; esac
        fi
    fi

    # 确认
    echo
    if [[ "$KEEP_OLD" == "yes" ]]; then
        warn "将把 SSH 端口改为 ${BOLD}${NEW_PORT}${NC}，并临时保留旧端口 ${BOLD}${old_port}${NC}。"
    else
        warn "将把 SSH 端口改为 ${BOLD}${NEW_PORT}${NC}（不保留旧端口）。"
    fi
    if [[ $ASSUME_YES -ne 1 && $DRY_RUN -ne 1 ]]; then
        c="$(prompt "$(echo -e "${CYAN}确认执行吗？(y/N) ${NC}")")" || c=""
        [[ "$c" == [yY] ]] || { info "已取消操作。"; exit 0; }
    fi

    # 备份
    if [[ $DRY_RUN -ne 1 ]]; then
        mkdir -p "$BACKUP_DIR"
        BACKUP_FILE="${BACKUP_DIR}/sshd_config.$(date +%Y%m%d-%H%M%S).bak"
        cp -p "$SSHD_CONFIG" "$BACKUP_FILE"
        ok "已备份原配置：$BACKUP_FILE"
        log "backup: $BACKUP_FILE"
    fi

    # Include 检查
    [[ $DRY_RUN -ne 1 ]] && handle_includes

    # 写入配置
    apply_config "$NEW_PORT" "$old_port" "$KEEP_OLD"

    if [[ $DRY_RUN -eq 1 ]]; then
        ok "预演完成，未修改任何文件。"
        exit 0
    fi

    # 语法校验
    info "校验配置语法 (sshd -t) ..."
    if ! sshd -t; then
        err "配置语法校验失败，回滚 ..."
        cp -p "$BACKUP_FILE" "$SSHD_CONFIG"
        CHANGED=0
        restart_ssh || true
        err "已恢复原配置。"
        log "sshd -t failed, rolled back"
        exit 1
    fi
    ok "语法校验通过。"

    # 防火墙
    configure_firewall "$NEW_PORT"

    # 重启
    info "重启 ssh 服务 ..."
    if restart_ssh; then
        ok "ssh 服务已重启。"
    else
        err "重启失败，请手动检查（当前会话先别断）。"
    fi

    # 验证监听
    sleep 1
    local listening=0
    for _ in 1 2 3 4 5; do
        if port_listening "$NEW_PORT"; then listening=1; break; fi
        sleep 1
    done
    if [[ $listening -eq 1 ]]; then
        ok "已验证端口 ${NEW_PORT} 处于监听状态。"
    else
        warn "未检测到端口 ${NEW_PORT} 监听，请执行：ss -tlnp | grep sshd"
    fi

    # 收尾提示
    echo
    echo -e "${BOLD}==================================================${NC}"
    echo -e "${BOLD}               重要安全提示${NC}"
    echo -e "${BOLD}==================================================${NC}"
    echo -e "${YELLOW}1.${NC} 请【新开一个终端】用新端口测试登录："
    echo -e "     ssh -p ${BOLD}${NEW_PORT}${NC} 用户@服务器IP"
    echo -e "${YELLOW}2.${NC} 确认能登录后，再断开当前会话。"
    if [[ "$KEEP_OLD" == "yes" ]]; then
        echo -e "${YELLOW}3.${NC} 验证通过后运行收尾脚本关闭旧端口："
        echo -e "     sudo bash remove_old_ssh_port.sh"
    fi
    echo -e "${YELLOW}4.${NC} 如需回滚："
    echo -e "     cp ${BACKUP_FILE} ${SSHD_CONFIG} && systemctl restart ssh"
    echo -e "${BOLD}==================================================${NC}"

    log "done: new_port=$NEW_PORT keep_old=$KEEP_OLD listening=$listening"
    ok "执行完毕。"
}

main "$@"

#!/usr/bin/env bash
#
# remove_old_ssh_port.sh —— 过渡期结束后，安全移除旧 SSH 端口的收尾脚本 (v2.8)
#
# 使用场景：
#   在 change_ssh_port.sh 中选择「临时保留旧端口」后，确认新端口登录正常，
#   运行本脚本把旧端口彻底关闭。
#
# v2 改进：
#   + 支持 -p 指定要移除的端口、-y 自动确认、-n 预演
#   + 备份 + sshd -t 校验 + 异常自动回滚
#   + 重启后验证剩余端口仍在监听
#
# 用法：
#   sudo bash remove_old_ssh_port.sh            # 交互
#   sudo bash remove_old_ssh_port.sh -p 22      # 指定移除端口
#   sudo bash remove_old_ssh_port.sh -p 22 -y   # 全自动

set -euo pipefail

readonly SCRIPT_VERSION="2.8"
readonly SSHD_CONFIG="/etc/ssh/sshd_config"
readonly BACKUP_DIR="/etc/ssh/sshd_config_backups"
readonly LOG_FILE="/var/log/change_ssh_port.log"

RED=$'\033[0;31m'; GREEN=$'\033[0;32m'; YELLOW=$'\033[1;33m'
CYAN=$'\033[0;36m'; BOLD=$'\033[1m'; NC=$'\033[0m'

info() { echo -e "${CYAN}[信息]${NC} $*"; }
ok()   { echo -e "${GREEN}[完成]${NC} $*"; }
warn() { echo -e "${YELLOW}[警告]${NC} $*"; }
err()  { echo -e "${RED}[错误]${NC} $*"; }

# 交互读取：stdin 不是终端时（如 curl | bash 管道执行）改从 /dev/tty 读取
# 用法：ans="$(prompt "提示文字")" || ans=""
# 用子 shell 探测 /dev/tty 是否真可用，避免 "No such device or address" 噪音
_tty_ok() { { : < /dev/tty; } 2>/dev/null; }

prompt() {
    local __text="$1" __ans=""
    # 提示必须单独写到 stderr。
    # 不能用 { read -p ...; } 2>/dev/null：read -p 的提示本身走 stderr，
    # 会被 2>/dev/null 一起吞掉，表现为"光标停住但没有任何提示"。
    if [[ -t 0 ]]; then
        printf '%s' "$__text" >&2
        read -r __ans
    elif _tty_ok; then
        printf '%s' "$__text" >&2
        read -r __ans < /dev/tty 2>/dev/null || return 1
    else
        return 1
    fi
    printf '%s' "$__ans"
}

log() {
    local ts; ts="$(date '+%F %T')"

    systemctl restart ssh  2>/dev/null && return 0
    systemctl restart sshd 2>/dev/null && return 0
    service ssh restart    2>/dev/null && return 0
    service sshd restart   2>/dev/null && return 0
    return 1
}

get_ports() {
    # 去重 + 数字排序，避免重复 Port 行导致"端口数量"判断出错
    grep -E '^[[:space:]]*Port[[:space:]]+' "$SSHD_CONFIG" 2>/dev/null \
        | awk '{print $2}' | sort -un || true
}

port_listening() {
    local port="$1"
    # 第 4 列为 Local Address:Port；表头该列不含端口号，故无需跳过首行
    if command -v ss >/dev/null 2>&1; then
        ss -tln 2>/dev/null | awk '{print $4}' | grep -qE "[:.]${port}\$"
    elif command -v netstat >/dev/null 2>&1; then
        netstat -tln 2>/dev/null | awk '{print $4}' | grep -qE "[:.]${port}\$"
    else
        return 1
    fi
}

CHANGED=0

rollback_if_needed() {
    local rc=$?
    if [[ $rc -ne 0 && $CHANGED -eq 1 && -n "$BACKUP_FILE" && -f "$BACKUP_FILE" ]]; then
        err "检测到异常，正在回滚 ..."
        cp -p "$BACKUP_FILE" "$SSHD_CONFIG" 2>/dev/null || true

    mapfile -t ports < <(get_ports)

    if [[ ${#ports[@]} -eq 0 ]]; then
        err "配置中未找到 Port 行（可能使用默认端口 22），无需移除。"
        exit 1
    fi

    info "当前生效端口：${BOLD}${ports[*]}${NC}"

    if [[ ${#ports[@]} -eq 1 ]]; then
        info "当前只有一个端口（${ports[0]}），没有可移除的旧端口。"
        exit 0
    fi

    # 非交互环境且未指定端口时，直接提示，避免 read 空转
    if [[ -z "$TARGET" && ! -t 0 ]] && ! _tty_ok; then
        err "当前为非交互环境，请用 -p 指定要移除的端口，例如：bash $0 -p 22 -y"

        if ! printf '%s\n' "${ports[@]}" | grep -qx "$TARGET"; then

        exit 1

        exit 0

        exit 1

        ok "ssh 服务已重启。"

        fi

    log "removed port $TARGET, remain=$(get_ports | tr '\n' ' ')"
}

# ---------- 工具函数 ----------
sha256_of() {
    if command -v sha256sum >/dev/null 2>&1; then
        sha256sum "$1" | awk '{print $1}'
    elif command -v shasum >/dev/null 2>&1; then
        shasum -a 256 "$1" | awk '{print $1}'
    else
        return 1
    fi
}

decode_payload() {
    local dest="$1"
    # 优先 GNU base64 -d，失败则回退 BSD base64 -D
    if printf '%s' "$PAYLOAD_B64" | base64 -d 2>/dev/null | tar -xzf - -C "$dest" 2>/dev/null; then
        return 0
    fi
    printf '%s' "$PAYLOAD_B64" | base64 -D 2>/dev/null | tar -xzf - -C "$dest" 2>/dev/null
}

# 创建临时目录（仅面向 Linux：mktemp -d 返回标准 POSIX 路径）
make_tmpdir() {
    mktemp -d
}

verify_payload() {
    local dest="$1" line sum name actual fail=0
    while IFS= read -r line; do
        if [[ -z "$line" ]]; then
            continue
        fi
        sum="${line%%  *}"
        name="${line#*  }"
        if ! actual="$(sha256_of "$dest/$name")"; then
            warn "缺少 sha256sum/shasum，跳过完整性校验"
            return 0
        fi
        if [[ "$sum" != "$actual" ]]; then
            err "校验失败: $name"
            fail=1
        fi
    done <<< "$MANIFEST"
    return "$fail"
}

# ---------- 别名注入 ----------
ALIAS_BEGIN="# >>> ssh-port-tool >>>"
ALIAS_END="# <<< ssh-port-tool <<<"

# alias 块内容：使用 sshh 作为入口，不覆盖系统自带的 ssh 命令
alias_block() {
    cat <<EOF
${ALIAS_BEGIN}
# Debian SSH 端口管理工具：输入 sshh 打开交互面板
# 刻意不占用 ssh，避免与系统命令冲突；ssh 仍为原生客户端，行为不变
# 如需直接运行面板而不依赖别名：ssh-port
alias sshh='ssh-port'
${ALIAS_END}
EOF
}

# 解析需要写入别名的用户 home 目录。
# 关键：sudo 下 $HOME 是 /root，而 $SUDO_USER 是"用户名"不是路径，
# 直接把它当目录用会导致普通用户的 rc 被静默跳过（装完 sshh 不生效）。
resolve_user_home() {
    local user="${SUDO_USER:-}" home=""
    if [[ -n "$user" && "$user" != "root" ]]; then
        if command -v getent >/dev/null 2>&1; then
            home="$(getent passwd "$user" 2>/dev/null | awk -F: '{print $6; exit}')" || home=""
        fi
        if [[ -z "$home" && -r /etc/passwd ]]; then
            home="$(awk -F: -v u="$user" '$1==u{print $6; exit}' /etc/passwd 2>/dev/null)" || home=""
        fi
    fi
    # 解析失败或目录不存在时退回当前 $HOME
    if [[ -z "$home" || ! -d "$home" ]]; then
        home="${HOME:-/root}"
    fi
    printf '%s' "$home"
}

# 向指定 rc 文件注入 alias 块（幂等：已存在则先移除旧块）
inject_alias() {
    local rc="$1"
    [[ -n "$rc" ]] || return 0
    # 只处理真实存在的 rc 文件，避免凭空创建 .bashrc
    [[ -f "$rc" ]] || return 0

    # 记录原属主：sudo 下 sed -i 会重建文件，属主会变成 root，
    # 导致用户之后无法编辑自己的 .bashrc
    local owner group
    owner="$(stat -c '%u' "$rc" 2>/dev/null)" || owner=""
    group="$(stat -c '%g' "$rc" 2>/dev/null)" || group=""

    # 已存在旧块 → 先删除，保证幂等
    if grep -qF "$ALIAS_BEGIN" "$rc" 2>/dev/null; then
        sed -i "\|^${ALIAS_BEGIN}\$|,\|^${ALIAS_END}\$|d" "$rc" 2>/dev/null || true
    fi
    printf '\n%s\n' "$(alias_block)" >> "$rc"

    if [[ -n "$owner" && -n "$group" && $EUID -eq 0 ]]; then
        chown "$owner:$group" "$rc" 2>/dev/null || true
    fi
}

remove_alias() {
    local rc
    for rc in "$@"; do
        [[ -f "$rc" ]] || continue
        if grep -qF "$ALIAS_BEGIN" "$rc" 2>/dev/null; then
            local owner group
            owner="$(stat -c '%u' "$rc" 2>/dev/null)" || owner=""
            group="$(stat -c '%g' "$rc" 2>/dev/null)" || group=""
            sed -i "\|^${ALIAS_BEGIN}\$|,\|^${ALIAS_END}\$|d" "$rc" 2>/dev/null || true
            if [[ -n "$owner" && -n "$group" && $EUID -eq 0 ]]; then
                chown "$owner:$group" "$rc" 2>/dev/null || true
            fi
            ok "已清理别名配置：$rc"
        fi
    done
}

install_alias() {
    local target_home
    target_home="$(resolve_user_home)"

    # 当前登录用户 + root，各家 shell 配置都处理
    local homes=() rc
    homes+=("$target_home")
    if [[ "$target_home" != "/root" && -d /root ]]; then
        homes+=("/root")
    fi

    local home
    for home in "${homes[@]}"; do
        [[ -d "$home" ]] || continue
        for rc in "$home/.bashrc" "$home/.zshrc"; do
            inject_alias "$rc"
        done
    done

    ok "已注入 sshh 别名到 ${target_home}（重新打开终端或执行 source ~/.bashrc 生效）"
}

# ---------- 安装 / 卸载 ----------
# 全局保存临时目录，便于 exec 前主动清理（exec 会跳过 EXIT trap）
TMP_INSTALL=""

do_install() {
    [[ -n "$DIR" && "$DIR" != "/" ]] || die "安装目录非法: '$DIR'"
    mkdir -p "$DIR" || die "无法创建目录 $DIR"

    TMP_INSTALL="$(make_tmpdir)" || die "无法创建临时目录"
    trap 'rm -rf "${TMP_INSTALL:-}"' EXIT

    info "解压安装包 ..."
    decode_payload "$TMP_INSTALL" || die "解压失败（base64/tar 不可用？）"

    # 解压结果自检：核心脚本必须到位
    local core
    for core in change_ssh_port.sh remove_old_ssh_port.sh ssh_menu.sh ssh-port.sh; do
        [[ -f "$TMP_INSTALL/$core" ]] || die "解压结果不完整，缺少 $core"
    done

    info "校验文件完整性 ..."
    verify_payload "$TMP_INSTALL" || die "完整性校验失败，已中止"

    info "安装到 $DIR ..."
    cp -f "$TMP_INSTALL/change_ssh_port.sh"     "$DIR/" || die "复制 change_ssh_port.sh 失败"
    cp -f "$TMP_INSTALL/remove_old_ssh_port.sh" "$DIR/" || die "复制 remove_old_ssh_port.sh 失败"
    cp -f "$TMP_INSTALL/ssh_menu.sh"            "$DIR/" || die "复制 ssh_menu.sh 失败"
    cp -f "$TMP_INSTALL/ssh-port.sh"            "$DIR/" || die "复制 ssh-port.sh 失败"
    cp -f "$TMP_INSTALL/check.sh"               "$DIR/" || die "复制 check.sh 失败"
    cp -f "$TMP_INSTALL/.shellcheckrc"          "$DIR/" 2>/dev/null || true
    chmod 755 "$DIR/change_ssh_port.sh" "$DIR/remove_old_ssh_port.sh" \
             "$DIR/ssh_menu.sh" "$DIR/ssh-port.sh" "$DIR/check.sh"

    if [[ $NO_DOC -eq 0 ]]; then
        cp -f "$TMP_INSTALL/README.md"   "$DIR/" 2>/dev/null || true
        cp -f "$TMP_INSTALL/使用文档.md" "$DIR/" 2>/dev/null || true
        chmod 644 "$DIR"/*.md 2>/dev/null || true
    fi

    # 命令软链
    if [[ -d /usr/local/bin && -w /usr/local/bin ]]; then
        ln -sf "$DIR/change_ssh_port.sh"     /usr/local/bin/ssh-port-change
        ln -sf "$DIR/remove_old_ssh_port.sh" /usr/local/bin/ssh-port-remove
        ln -sf "$DIR/ssh-port.sh"            /usr/local/bin/ssh-port
        ln -sf "$DIR/ssh_menu.sh"            /usr/local/bin/ssh-port-menu
        ok "已创建命令：ssh-port / ssh-port-menu / ssh-port-change / ssh-port-remove"
    else
        warn "未创建 /usr/local/bin 软链（目录不可写）"
    fi

    # 别名注入
    if [[ $NO_ALIAS -eq 0 ]]; then
        install_alias
    else
        info "已跳过 sshh 别名注入（--no-alias）"
    fi

    ok "安装完成：$DIR"
    echo
    echo -e "${BOLD}接下来怎么用：${NC}"
    echo -e "  ${GREEN}sshh${NC}                  打开交互式管理面板（推荐）"
    echo -e "  ${GREEN}ssh-port-menu${NC}         同上，直接命令"
    echo -e "  ${GREEN}sudo ssh-port-change${NC}  直接更换端口"
    echo
    echo -e "${DIM}提示：重新打开一个终端，或执行 source ~/.bashrc 让 sshh 生效。${NC}"
    echo -e "${DIM}      系统自带的 ssh 命令不受影响：ssh user@host 仍是原生客户端。${NC}"
    echo -e "${DIM}      若之前装过旧版（占用过 ssh），当前终端可执行 unalias ssh 清除残留。${NC}"
}

do_uninstall() {
    [[ -n "$DIR" && "$DIR" != "/" ]] || die "安装目录非法: '$DIR'"
    rm -f /usr/local/bin/ssh-port /usr/local/bin/ssh-port-menu \
          /usr/local/bin/ssh-port-change /usr/local/bin/ssh-port-remove

    local target_home
    target_home="$(resolve_user_home)"
    local homes=() rc
    homes+=("$target_home")
    if [[ "$target_home" != "/root" && -d /root ]]; then
        homes+=("/root")
    fi
    local home
    for home in "${homes[@]}"; do
        [[ -d "$home" ]] || continue
        for rc in "$home/.bashrc" "$home/.zshrc"; do
            remove_alias "$rc"
        done
    done

    if [[ -d "$DIR" ]]; then
        rm -rf "$DIR"
        ok "已删除 $DIR"
    else
        info "目录不存在：$DIR"
    fi
    ok "卸载完成"
}

# ---------- 主流程 ----------
main() {
    echo -e "${BOLD}==================================================${NC}"
    echo -e "${BOLD}   SSH 端口更换工具 · 安装器 v${VERSION}${NC}"
    echo -e "${BOLD}==================================================${NC}"
    echo

    [[ $EUID -eq 0 ]] || die "请使用 root 运行（sudo bash install.sh）"
    [[ "$(uname -s)" == "Linux" ]] || die "本工具仅支持 Linux（Debian/Ubuntu 等）"
    command -v tar    >/dev/null 2>&1 || die "缺少 tar"
    command -v base64 >/dev/null 2>&1 || die "缺少 base64"

    if [[ $DO_UNINSTALL -eq 1 ]]; then
        do_uninstall
        return 0
    fi

    if [[ $ASSUME_YES -ne 1 ]]; then
        local ans
        ans="$(prompt "将安装到 ${DIR}，继续？(Y/n) ")" || ans=""
        case "$ans" in
            [nN]) info "已取消"; exit 0 ;;
        esac
    fi

    do_install

    # exec 会替换进程，EXIT trap 不再触发，这里先主动清理临时目录
    if [[ $MENU_AFTER -eq 1 || $RUN_AFTER -eq 1 ]]; then
        rm -rf "${TMP_INSTALL:-}" 2>/dev/null || true
        TMP_INSTALL=""
        trap - EXIT
    fi

    if [[ $MENU_AFTER -eq 1 ]]; then
        echo
        info "启动交互式管理面板 ..."
        exec bash "$DIR/ssh_menu.sh" --menu
    fi

    if [[ $RUN_AFTER -eq 1 ]]; then
        echo
        info "启动更换端口脚本 ..."
        exec bash "$DIR/change_ssh_port.sh"
    fi
}

main "$@"

#!/usr/bin/env bash
#
# push-to-github.sh — 将本项目一键推送到 GitHub
#
# 设计要点：
#   1. Token 使用 read -s 静默读取，不落盘、不进 .git/config
#   2. 推送时通过一次性 credential.helper 传入 Token（不出现在进程列表）
#   3. 兼容 curl | bash（stdin 被占用时自动从 /dev/tty 读取）
#   4. 支持非交互：GH_TOKEN / GH_USER / GH_REPO 环境变量 + 参数
#
# 用法：
#   bash push-to-github.sh                    # 全交互
#   bash push-to-github.sh -u yourname -r myrepo --private
#
set -Eeuo pipefail

readonly SCRIPT_VERSION="1.0"

RED=$'\033[0;31m'
GREEN=$'\033[0;32m'
YELLOW=$'\033[1;33m'
CYAN=$'\033[0;36m'
BOLD=$'\033[1m'
NC=$'\033[0m'

info() { printf '%s\n' "${CYAN}[信息]${NC} $*"; }
ok()   { printf '%s\n' "${GREEN}[成功]${NC} $*"; }
warn() { printf '%s\n' "${YELLOW}[警告]${NC} $*"; }
err()  { printf '%s\n' "${RED}[错误]${NC} $*" >&2; }
die()  { err "$*"; exit 1; }

# ---------------------------------------------------------------- 交互读取
# stdin 不是终端（curl | bash）时改从 /dev/tty 读，否则 read 会立刻 EOF
# 用子 shell 探测 /dev/tty 是否真的可用，避免 "No such device or address" 噪音
_tty_ok() { { : < /dev/tty; } 2>/dev/null; }

prompt() {
    local text="$1" ans=""
    if [[ -t 0 ]]; then
        read -r -p "$text" ans || ans=""
    elif _tty_ok; then
        { read -r -p "$text" ans < /dev/tty; } 2>/dev/null || ans=""
    else
        return 1
    fi
    printf '%s' "$ans"
}

# 静默读取（Token / 密码），不回显
prompt_secret() {
    local text="$1" ans=""
    if [[ -t 0 ]]; then
        read -r -s -p "$text" ans || ans=""
        printf '\n' >&2
    elif _tty_ok; then
        { read -r -s -p "$text" ans < /dev/tty; } 2>/dev/null || ans=""
        printf '\n' >&2
    else
        return 1
    fi
    printf '%s' "$ans"
}

# ---------------------------------------------------------------- 参数
GH_USER="${GH_USER:-}"
GH_REPO="${GH_REPO:-}"
GH_TOKEN="${GH_TOKEN:-${GITHUB_TOKEN:-}}"
VISIBILITY="private"
MODE="https"          # https | ssh | local
DRY_RUN=0
ASSUME_YES=0

usage() {
    cat <<'EOF'
push-to-github.sh — 将本项目推送到 GitHub

用法:
  bash push-to-github.sh [选项]

选项:
  -u, --user <name>    GitHub 用户名
  -r, --repo <name>    仓库名（默认 debian-ssh-port-tool）
      --public         创建公开仓库（默认 private）
      --private        创建私有仓库（默认）
      --https          使用 HTTPS + 访问令牌推送（默认，推荐）
      --ssh            使用 SSH 推送（需已配置密钥，且 22 端口可达）
      --local          只做本地提交，不推送，由你手动推
  -t, --token <tok>    访问令牌（建议改用 GH_TOKEN 环境变量，避免留在 shell 历史）
  -y, --yes            跳过最终确认
  -n, --dry-run        预演，只显示将执行的命令
  -V, --version        显示版本
  -h, --help           显示本帮助

环境变量:
  GH_TOKEN / GITHUB_TOKEN   访问令牌
  GH_USER                    GitHub 用户名
  GH_REPO                    仓库名

Token 权限: 勾选 repo（私有仓库必需）。创建 Token:
  https://github.com/settings/tokens/new
EOF
}

while [[ $# -gt 0 ]]; do
    case "$1" in
        -u|--user)
            [[ $# -ge 2 ]] || die "选项 $1 缺少参数"
            GH_USER="$2"; shift 2 ;;
        -r|--repo)
            [[ $# -ge 2 ]] || die "选项 $1 缺少参数"
            GH_REPO="$2"; shift 2 ;;
        -t|--token)
            [[ $# -ge 2 ]] || die "选项 $1 缺少参数"
            GH_TOKEN="$2"; shift 2 ;;
        --public)   VISIBILITY="public"; shift ;;
        --private)  VISIBILITY="private"; shift ;;
        --https)    MODE="https"; shift ;;
        --ssh)      MODE="ssh"; shift ;;
        --local)    MODE="local"; shift ;;
        -y|--yes)   ASSUME_YES=1; shift ;;
        -n|--dry-run) DRY_RUN=1; shift ;;
        -V|--version) printf 'push-to-github.sh v%s\n' "$SCRIPT_VERSION"; exit 0 ;;
        -h|--help)  usage; exit 0 ;;
        *) die "未知选项: $1（用 -h 查看帮助）" ;;
    esac
done

# ---------------------------------------------------------------- 工具函数
run() {
    if [[ $DRY_RUN -eq 1 ]]; then
        printf '%s\n' "${YELLOW}[预演]${NC} $*"
    else
        "$@"
    fi
}

# 用一次性 credential.helper 把 Token 传给 git，不写入 .git/config
git_push_with_token() {
    local url="$1" branch="$2"
    # 单引号是故意的：$GH_USER / $GH_TOKEN 需由 git 调用 helper 时才展开
    # shellcheck disable=SC2016
    GH_USER="$GH_USER" GH_TOKEN="$GH_TOKEN" \
        git -c credential.helper='!f() { echo username="$GH_USER"; echo password="$GH_TOKEN"; }; f' \
        push -u "$url" "$branch"
}

# 拉取远程 main；远程为空仓库时返回非零（调用方容忍即可）
git_fetch_remote() {
    local url="$1"
    if [[ "$MODE" == "https" ]]; then
        # shellcheck disable=SC2016
        GH_USER="$GH_USER" GH_TOKEN="$GH_TOKEN" GIT_TERMINAL_PROMPT=0 \
            git -c credential.helper='!f() { echo username="$GH_USER"; echo password="$GH_TOKEN"; }; f' \
            fetch -q "$url" main >/dev/null 2>&1
    else
        GIT_TERMINAL_PROMPT=0 git fetch -q "$url" main >/dev/null 2>&1
    fi
}

# 调 GitHub API 创建仓库（Token 通过 curl 配置文件传入，不进 argv）
create_repo_api() {
    local name="$1" vis="$2" cfg rc=0
    cfg="$(mktemp)"
    chmod 600 "$cfg"
    {
        printf 'header = "Authorization: Bearer %s"\n' "$GH_TOKEN"
        printf 'header = "Accept: application/vnd.github+json"\n'
        printf 'header = "X-GitHub-Api-Version: 2022-11-28"\n'
    } > "$cfg"

    local payload
    payload="$(printf '{"name":"%s","private":%s,"description":"Debian SSH 端口更换工具（交互式 / 自解压安装）"}' \
        "$name" "$([[ "$vis" == "private" ]] && echo true || echo false)")"

    local out
    out="$(curl -sS -m 30 -w '\n%{http_code}' -X POST -K "$cfg" \
        https://api.github.com/user/repos -d "$payload" 2>&1)" || rc=$?
    rm -f "$cfg"

    if [[ $rc -ne 0 ]]; then
        err "调用 GitHub API 失败（网络问题？）"
        printf '%s\n' "$out"
        return 1
    fi

    local code
    code="$(printf '%s' "$out" | tail -n1)"
    case "$code" in
        201)
            ok "已在 GitHub 创建仓库: $GH_USER/$name（$vis）"
            return 0 ;;
        422)
            warn "仓库 $GH_USER/$name 已存在，继续推送。"
            return 0 ;;
        401)
            err "Token 无效或已过期（HTTP 401）。请重新生成 Token。"
            return 1 ;;
        403)
            err "Token 权限不足（HTTP 403）。请确认已勾选 repo 权限。"
            return 1 ;;
        *)
            err "创建仓库失败，HTTP $code"
            printf '%s\n' "$out" | head -n -1
            return 1 ;;
    esac
}

# ---------------------------------------------------------------- 主流程
main() {
    printf '%s\n' "${BOLD}================================================${NC}"
    printf '%s\n' "${BOLD}  Debian SSH 端口更换工具 — 推送到 GitHub${NC}"
    printf '%s\n' "${BOLD}================================================${NC}"
    printf '\n'

    command -v git >/dev/null 2>&1 || die "未找到 git，请先安装：https://git-scm.com/downloads"

    local here root
    here="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)" || die "无法进入脚本所在目录"
    # 脚本可能位于分类子目录（如 Debian SSH/），需定位到仓库根再操作，
    # 否则 git add -A 只会暂存当前子目录
    root="$(git -C "$here" rev-parse --show-toplevel 2>/dev/null || true)"
    cd "${root:-$here}" || die "无法进入项目目录"
    root="$(pwd)"
    info "项目目录: $root"

    if ! git rev-parse --is-inside-work-tree >/dev/null 2>&1; then
        info "初始化 git 仓库 ..."
        run git init -b main >/dev/null
    fi

    # ---------- 提交身份 ----------
    local cname cemail
    cname="$(git config --get user.name 2>/dev/null || true)"
    cemail="$(git config --get user.email 2>/dev/null || true)"
    if [[ -z "$cname" || -z "$cemail" ]]; then
        warn "git 提交身份未配置，需要设置 user.name / user.email"
        [[ -n "$cname" ]] || cname="$(prompt "  Git 用户名: ")" || cname=""
        [[ -n "$cemail" ]] || cemail="$(prompt "  Git 邮箱（建议与 GitHub 一致）: ")" || cemail=""
        if [[ -z "$cname" || -z "$cemail" ]]; then
            die "无法读取提交身份（当前环境无终端）。请先手动设置：
  git config --local user.name \"你的名字\"
  git config --local user.email \"你的邮箱\""
        fi
        run git config --local user.name "$cname"
        run git config --local user.email "$cemail"
        ok "已设置本仓库提交身份: $cname <$cemail>"
    else
        info "提交身份: $cname <$cemail>"
    fi

    # ---------- 远程信息 ----------
    if [[ "$MODE" != "local" ]]; then
        if [[ -z "$GH_USER" ]]; then
            GH_USER="$(prompt "GitHub 用户名: ")" || GH_USER=""
            [[ -n "$GH_USER" ]] || die "未提供用户名（当前环境无终端，可用 -u 参数指定）"
        fi
        if [[ -z "$GH_REPO" ]]; then
            local def="debian-ssh-port-tool"
            local input
            input="$(prompt "仓库名（默认 $def）: ")" || input=""
            GH_REPO="${input:-$def}"
        fi
        printf '\n'
        info "目标仓库: https://github.com/${GH_USER}/${GH_REPO}  （${VISIBILITY}）"
        info "推送方式: ${MODE}"

        # ---------- 认证准备 ----------
        if [[ "$MODE" == "https" && -z "$GH_TOKEN" ]]; then
            warn "需要一个 GitHub 访问令牌（Personal Access Token）"
            info "生成地址: https://github.com/settings/tokens/new  → 勾选 repo 权限"
            GH_TOKEN="$(prompt_secret "  粘贴 Token（输入不回显）: ")" || GH_TOKEN=""
            [[ -n "$GH_TOKEN" ]] || die "未提供 Token，已中止（可用 GH_TOKEN 环境变量或 -t 参数传入）"
        fi
        if [[ "$MODE" == "ssh" ]]; then
            warn "SSH 方式需要已配置密钥且 github.com:22 可达"
            info "检测到本机 SSH 22 端口常被网络阻断，若失败请改用 HTTPS"
        fi
        printf '\n'
    fi

    # ---------- 建仓与 remote（放在提交之前，便于对齐远程已有历史）----------
    local url=""
    if [[ "$MODE" != "local" ]]; then
        if [[ "$MODE" == "https" && $DRY_RUN -eq 0 ]]; then
            info "检查 / 创建远程仓库 ..."
            create_repo_api "$GH_REPO" "$VISIBILITY" || warn "跳过自动建仓，将直接尝试推送（仓库需已存在）"
        fi

        if [[ "$MODE" == "ssh" ]]; then
            url="git@github.com:${GH_USER}/${GH_REPO}.git"
        else
            url="https://github.com/${GH_USER}/${GH_REPO}.git"
        fi

        if git remote get-url origin >/dev/null 2>&1; then
            info "更新 origin → $url"
            run git remote set-url origin "$url"
        else
            info "添加 origin → $url"
            run git remote add origin "$url"
        fi

        # 远程若已有初始提交（建仓时勾了 README 就会这样），把本地接到它上面，
        # 避免 unrelated histories / non-fast-forward 报错。
        # 仅在本地还没有任何提交时执行，绝不会覆盖已有历史。
        if [[ $DRY_RUN -eq 0 ]] && ! git rev-parse --verify HEAD >/dev/null 2>&1; then
            if git_fetch_remote "$url"; then
                if git rev-parse --verify origin/main >/dev/null 2>&1; then
                    git reset --soft origin/main
                    ok "已对齐远程 main 历史（本次提交将作为它的后继）"
                fi
            fi
        fi
    fi

    # ---------- 提交 ----------
    run git add -A

    if git diff --cached --quiet 2>/dev/null; then
        info "没有需要提交的改动"
    else
        local msg="feat: Debian SSH 端口更换工具 v2.4

- change_ssh_port.sh: 交互式更换端口，保留旧端口过渡
- remove_old_ssh_port.sh: 验证通过后移除旧端口
- install.sh: 自解压一键安装（curl | bash）
- push-to-github.sh: 一键推送到 GitHub
- check.sh + GitHub Actions: ShellCheck CI
- 完整使用文档与高亮展示页"
        info "创建提交 ..."
        run git commit -m "$msg"
        ok "提交完成"
    fi

    run git branch -M main

    if [[ "$MODE" == "local" ]]; then
        printf '\n'
        ok "本地仓库已就绪（未推送）"
        printf '\n后续手动推送：\n'
        printf '  git remote add origin https://github.com/<你的用户名>/<仓库名>.git\n'
        printf '  git push -u origin main\n'
        return 0
    fi

    # ---------- 最终确认 ----------
    if [[ $ASSUME_YES -eq 0 && $DRY_RUN -eq 0 ]]; then
        printf '\n'
        local c
        c="$(prompt "确认推送到 https://github.com/${GH_USER}/${GH_REPO} ？(y/N) ")" || c=""
        case "$c" in
            y|Y|yes|YES) ;;
            *) warn "已取消。"; exit 0 ;;
        esac
    fi

    # ---------- 推送 ----------
    printf '\n'
    info "推送中 ..."
    local rc=0
    if [[ $DRY_RUN -eq 1 ]]; then
        run git push -u origin main
    elif [[ "$MODE" == "https" ]]; then
        git_push_with_token "$url" main || rc=$?
    else
        git push -u origin main || rc=$?
    fi

    printf '\n'
    if [[ $rc -ne 0 ]]; then
        err "推送失败（退出码 $rc）"
        printf '\n排查建议：\n'
        printf '  1. Token 是否勾选了 repo 权限、是否过期\n'
        printf '  2. 仓库名是否与 GitHub 上一致（大小写敏感）\n'
        printf '  3. 若报 non-fast-forward / unrelated histories，说明远程已有提交：\n'
        printf '     git fetch origin main && git log --oneline origin/main   # 先看清远程内容\n'
        printf '     git pull --rebase --allow-unrelated-histories origin main # 再合并\n'
        printf '  4. 网络受限可配置代理: git config --global http.proxy http://127.0.0.1:7890\n'
        exit "$rc"
    fi

    ok "推送成功！"
    printf '\n仓库地址: %s\n' "${GREEN}https://github.com/${GH_USER}/${GH_REPO}${NC}"
    printf '\n后续提交只需：\n'
    printf '  git add -A && git commit -m "说明" && git push\n'
}

main "$@"
